@RequestMapping(value = "/deleteCart", method = RequestMethod.POST)
	@RequireHardLogIn
	@ResponseBody
	public MultiCartData deleteMultiCart(@RequestBody MultiCartData multiCartData, final HttpServletRequest request,
			final RedirectAttributes redirectModel) throws CommerceSaveCartException {
		final Locale loc = i18nService.getCurrentLocale();
		boolean isDeleted = false;
		String cartNameToBeDeleted = multiCartData.getCartName();
		
		if (multiCartData.getIsActiveCart()) {
			// active cart delete
			boolean isRestored = false;
			boolean isCreated = false;
			boolean setCartNamePostDeletion = false;
			// get the current cart from session
			CartModel cartToDelete = cartService.getSessionCart();

			// remove the current cart from session
			sessionService.removeAttribute(SESSION_CART_PARAMETER_NAME);

			if (!multiCartData.getIsOnlyActiveCartAvailable()) {
				// saved carts exists for the user
				// restore a previously saved cart
				isRestored = multiCartFacade.restoreSavedCart(multiCartData.getCartIdToBeActive());
			} else {
				// no saved carts exists for the user
				final String newCartName = multiCartService.getNewCartName(cartToDelete);
				//CLOUD-52294 - flag created to set new unique cart name post deletion of active cart
				setCartNamePostDeletion = newCartName.equals(
						RexelUtils.getMessageSource().getMessage(RexelConstants.DEFAULT_CART_NAME, null, loc)) ? false
								: true;
				multiCartData.setCartName(newCartName);
				isCreated = multiCartFacade.createNewSavedCart(multiCartData);
			}
			
			if (isRestored || isCreated) {
				// delete the old session cart
				isDeleted = multiCartFacade.deleteSavedCart(cartToDelete.getCode());
				//CLOUD-52294 - set new unique cart name post deletion of active cart
				if (isCreated && setCartNamePostDeletion) {
					final CartModel sessionCart = cartService.getSessionCart();
					multiCartFacade.setNewCartName(sessionCart);
				}
			} else {
				// set the old cart to session
				cartService.setSessionCart(cartToDelete);
			}
		} else {
			// saved cart delete
			isDeleted = multiCartFacade.deleteSavedCart(multiCartData.getCartIdToBeDeleted());
		}
		
		if (isDeleted) {
			multiCartData.setSuccessMessage(cartNameToBeDeleted + " " + messageSource.getMessage(
					"delete.multiple.cart.on.success", null, "is successfully deleted from the cart list", loc));
			multiCartData.setSavedCartList(multiCartFacade
					.getSavedCartsForCurrentUser(baseSiteService.getCurrentBaseSite(), userService.getCurrentUser()));
		} else {
			multiCartData.setErrorMessage(messageSource.getMessage(
					"delete.restore.multiple.cart.on.error", null, "Unexpected error occurred. Please try again", loc));
		}

		return multiCartData;
	}

    private TimeService timeService;
	
	private static final String GET_CART_FOR_NAME_AND_USER_QUERY = "SELECT {pk} FROM {cart!} WHERE {user} = ?user AND {site} = ?site AND {name} = ?name AND {saveTime} IS NOT NULL AND {expirationTime} >= ?currentDate";
	
	/**
	 * This method is used to get the saved carts for the site and user
	 * 
	 * @param baseSite is BaseSiteModel type
	 * @param user is UserModel type
	 * @return List<CartModel>
	 */
	@Override
	public List<CartModel> getSavedCartsForSiteAndUser(final BaseSiteModel baseSite,
			final UserModel user)
	{
		final Map<String, Object> params = new HashMap<String, Object>();
        params.put("user", user);
        params.put("site", baseSite);
        params.put("currentDate", timeService.getCurrentTime());
        //It retrieves the primary keys ({pk}) of Cart instances where the user attribute matches the provided user, the site attribute matches the provided site, and the saveTime attribute is not null. The results are ordered in descending order by the saveTime attribute.
        final String query = "SELECT {pk} FROM {cart!} WHERE {user} = ?user AND {site} = ?site AND {saveTime} IS NOT NULL AND {expirationTime} >= ?currentDate ORDER BY LOWER({name}) ASC";
        final FlexibleSearchQuery fQuery = new FlexibleSearchQuery(query);
        fQuery.addQueryParameters(params);
        fQuery.setResultClassList(Arrays.asList(CartModel.class));
        final SearchResult<CartModel> searchResult = flexibleSearchService.search(fQuery);
        return searchResult.getResult();
	}
	
	/**
	 * @CloudLink:https://jira.rexel.com/browse/CLOUD-50202
	 * This method is used to get the cart with the given name for the site and user
	 * 
	 * @param cartName is String type
	 * @param baseSite is BaseSiteModel type
	 * @param user is UserModel type
	 * @return CartModel type
	 */
	@Override
	public CartModel getCartForNameAndUser(final String cartName, final BaseSiteModel baseSite, final UserModel user) {
		final Map<String, Object> params = new HashMap<String, Object>();
		params.put("user", user);
		params.put("site", baseSite);
		params.put("currentDate", timeService.getCurrentTime());
		params.put("name", cartName);
		// It retrieves the primary keys ({pk}) of Cart instances where the user attribute matches the provided user, the site attribute matches the provided site, the provided cart name and the saveTime attribute is not null.
		final String query = GET_CART_FOR_NAME_AND_USER_QUERY;
		final FlexibleSearchQuery fQuery = new FlexibleSearchQuery(query);
		fQuery.addQueryParameters(params);
		fQuery.setResultClassList(Arrays.asList(CartModel.class));
		final SearchResult<CartModel> searchResult = flexibleSearchService.search(fQuery);
		final List<CartModel> carts = searchResult.getResult();
		if (CollectionUtils.isNotEmpty(carts)) {
			return carts.get(0);
		}
		return null;
	}

}

		final List<CartModel> savedCartModels = multiCartService.getSavedCartsForSiteAndUser(baseSite,user);

			for (CartModel cartModel : savedCartModels) {
				MultiCartSavedList multiCartSaved = new MultiCartSavedList();
				multiCartSaved.setCartId(cartModel.getCode());
				multiCartSaved.setCartName(cartModel.getName());
				//CLOUD-49952- Set the cart entry size
				multiCartSaved.setCartEntrySize(cartModel.getEntries().size());

						.findFirst()
						.orElse(null);

				multiCartSavedList.add(multiCartSaved);

			}
		}
		return multiCartSavedList;

	/**
	 * This method is used to auto save the newly created non saved session cart to saved cart (having savedtime and name attributes updated)
	 */
	@Override
	public void autoSaveSessionCart() {
		// Get the current user and cast it to B2BCustomerModel
		final UserModel user = userService.getCurrentUser();
		final B2BCustomerModel customer = user instanceof B2BCustomerModel ? (B2BCustomerModel) user : null;
		//CLOUD-50228- Check if the user is a punchout user
		final boolean isPunchoutUser = rexelB2BCommerceUserFacade.isPunchoutUser(customer);
		if (rexelConfigService.getConfigBooleanValue(RexelConfigConstants.MULTIPLE_CARTS_FEATURE_ENABLE, false) && BooleanUtils.isFalse(isPunchoutUser) ) {
			final CartData cartData = rexelB2BCartFacade.getSessionCart();
			if (null == cartData.getSaveTime()) {
				final CommerceSaveCartParameterData commerceSaveCartParameterData = new CommerceSaveCartParameterData();
				// CLOUD-45035 - Code generated by Co-pilot -- Starts

				// Check if the cart should be persisted after an order is placed
				// Use BooleanUtils.isTrue() to handle null values and convert to boolean
				boolean isPersistCart = BooleanUtils.isTrue(customer.getPersistCartAfterPlaceOrder());

				// If the cart should be persisted, get the cart name from the session
				// Otherwise, set cartName to null
				String cartName = isPersistCart ? sessionService.getAttribute(ACTIVE_MULTI_CART_NAME) : null;

				// If the cart name is null or empty, set it to the default cart label
				if (StringUtils.isBlank(cartName)) {
					// CLOUD-50202 - get the new unique cart name
					cartName = multiCartService.getNewCartName(cartService.getSessionCart());
				}

				// Set the cart name in the commerceSaveCartParameterData object
				commerceSaveCartParameterData.setName(cartName);
				// Code generated by Co-pilot -- Ends
				commerceSaveCartParameterData.setCartId(cartData.getCode());
				commerceSaveCartParameterData.setEnableHooks(true);
	
				CommerceSaveCartResultData saveCartData = null;
				try {
					saveCartData = saveCartFacade.saveCart(commerceSaveCartParameterData);
					LOG.info("Converted to saved cart successfully for cart code: "
							+ saveCartData.getSavedCartData().getCode());
					sessionService.removeAttribute(ACTIVE_MULTI_CART_NAME);
				} catch (final CommerceSaveCartException e) {
					LOG.error("Saved cart conversion failed for cart code: " + cartData.getCode(), e);
				}
			}
		}

	/**CLOUD-49301
	 * This method is used to restore a saved cart given its cartId.
	 * @param cartId
	 * @return boolean
	 */
	@Override
	public boolean restoreSavedCart(String cartId) {
		try
		{
			final CommerceSaveCartParameterData commerceSaveCartParameterData = new CommerceSaveCartParameterData();
			commerceSaveCartParameterData.setCartId(cartId);
			commerceSaveCartParameterData.setEnableHooks(true);
			
			saveCartFacade.restoreSavedCart(commerceSaveCartParameterData);
			/**
			 * This code is used to update the modification time of a cart when it is restored. 
			 * The modification time is set to the current time, and then the cart is saved. 
			 * This ensures that the modification time of the cart reflects the time it was restored.
			 */
			final CartModel getrestoreCartForCodeAndUser= commerceCartService.getCartForCodeAndUser(cartId, userService.getCurrentUser());
			if (null != getrestoreCartForCodeAndUser) {
				getrestoreCartForCodeAndUser.setModifiedtime(timeService.getCurrentTime());
	            modelService.save(getrestoreCartForCodeAndUser);
	            modelService.refresh(getrestoreCartForCodeAndUser);
	            // CLOUD-50339 - Set the active cart id in the B2BCustomerModel
	            final B2BCustomerModel customer = (B2BCustomerModel) getrestoreCartForCodeAndUser.getUser();
				if (customer != null) {
					customer.setActiveCartId(getrestoreCartForCodeAndUser.getCode());
					modelService.save(customer);
					modelService.refresh(customer);
					LOG.info("Active cart Id: " + customer.getActiveCartId() + " for the user: " + customer.getUid());
				}
				// CLOUD-50671 - Clear the delivery address from the session when cart is restored
				if (rexelConfigService.getConfigBooleanValue(RexelConfigConstants.DELIVERYADDRESS_CACHING_ENABLED, null, true))
				{
					sessionService.removeAttribute(RexelConstants.CART_DELIVERY_ADDRESS);
				}
			}
		}

	 * CLOUD-50891-CLOUD-50981 
	 * It retrieves the current cart for the session of the customer and the saved cart list. 
	 * If the user does not want to persist the cart after placing an order and the cart is not null,
	 * it restores the cart based on the user's preference. If there are saved carts for the user and the session cart exists,
	 * it deletes the session cart.
	 */
	@Override
	public void processCartBasedOnUserPreference() {
		final CartModel cartModel = cartService.getSessionCart();
		final B2BCustomerModel customerModel = (B2BCustomerModel) userService.getCurrentUser();
		final boolean isPersistCart = customerModel != null ? BooleanUtils.isTrue(customerModel.getPersistCartAfterPlaceOrder()) : false;
		if (cartModel != null) {
			final CartModel removeSessionCartModel = cartModel;
			final List<MultiCartSavedList> savedCartList = getSavedCartsForCurrentUser(
					baseSiteService.getCurrentBaseSite(), userService.getCurrentUser());

			// CLOUD-53324 -- If the savedCartList contains the session cart created after placing an order, remove the session cart from the list when isPersistCart is false.
			if (BooleanUtils.isFalse(isPersistCart) && CollectionUtils.isNotEmpty(savedCartList)) {
				savedCartList.stream().anyMatch(cart -> cart.getCartId().equals(removeSessionCartModel.getCode()));
				removeSessionCartModel.setSaveTime(null);
			    modelService.save(removeSessionCartModel);
			    modelService.refresh(removeSessionCartModel);
			}
			restoreCartBasedOnUserPreference();
			final CartModel sessionCartModel = cartService.getSessionCart();
			// CLOUD-51259 -- If the user doesn't want to persist the cart after order placement, there are saved carts, and a session cart exists,
			// the session cart is logged and removed.
			if (BooleanUtils.isFalse(isPersistCart) && CollectionUtils.isNotEmpty(savedCartList)
					&& removeSessionCartModel != null && sessionCartModel != null && !sessionCartModel.equals(removeSessionCartModel)) {
				LOG.info("Deleted session cart id ----" + removeSessionCartModel.getCode());
				modelService.remove(removeSessionCartModel);
			}
		}
	}

	/**
	 * CLOUD-54321
	 * This method is used to move the products from session cart to selected cart
	 * @param multiCartData
	 * @throws FacetConfigServiceException, FacetSearchException
	 * @return status
	 */
	@Override
	public String moveProductsToSelectedCart(MultiCartData multiCartData) throws FacetConfigServiceException, FacetSearchException {
		String status = StringUtils.EMPTY;
		// Get the source cart and target carts in create new cart flow when IsEntriesMoveToNewCart is true else get the session cart and selected cart
		final CartModel sourceCart = BooleanUtils.isTrue(multiCartData.getIsEntriesMoveToNewCart()) 
		        ? rexelB2BCartFacade.getCartForCodeAndUser(multiCartData.getCartId()) 
		        : cartService.getSessionCart();
		final CartModel targetCart = BooleanUtils.isTrue(multiCartData.getIsEntriesMoveToNewCart()) 
		        ? cartService.getSessionCart() 
		        : rexelB2BCartFacade.getCartForCodeAndUser(multiCartData.getSelectedCartId());
		final int cartSizeLimit = rexelConfigService.getConfigIntValue(RexelConfigConstants.REXEL_CARTSIZE, null, 0);
		final int totalEntries = targetCart.getEntries().size() + multiCartData.getSelectedProductsToMove().size();

		if (totalEntries > cartSizeLimit) {
			return RexelConstants.CART_FULL;
		}
		//CLOUD-55351
		MultiCartData finalMultiCartData = multiCartData;
		final List<AbstractOrderEntryModel> toMoveEntries = sourceCart.getEntries().stream()
				.filter(orderEntry -> finalMultiCartData.getSelectedProductsToMove().contains(String.valueOf(orderEntry.getEntryNumber())))
				.collect(Collectors.toList());

		if (toMoveEntries.stream().anyMatch(entry -> entry.getEntryNumber() != null && entry.getFlagGroupingID() != null)) {
			final Map<Integer, Long> productEntriesToBeDeleted = targetCart.getEntries().stream()
					.filter(entry -> entry.getEntryNumber() != null && entry.getFlagGroupingID() != null)
					.collect(Collectors.toMap(AbstractOrderEntryModel::getEntryNumber, entry -> 0L));

			if (!productEntriesToBeDeleted.isEmpty()) {
				removeSelectedEntriesFromTargetCart(targetCart, productEntriesToBeDeleted);
			}
		}
		// CLOUD-49957 - Check if the products are from the same catalog
		final boolean orderingAcrossCatalogEnable = rexelConfigService.getConfigBooleanValue(RexelConfigConstants.CONFIG_ORDERING_ACROSS_CATALOGS_ENABLE, null, false);
		boolean movingProductOverCatalog = true;
		if (BooleanUtils.isFalse(orderingAcrossCatalogEnable)) {
			movingProductOverCatalog = isProductsFromSameCatalog(targetCart, multiCartData, true);
		}
		if (BooleanUtils.isTrue(movingProductOverCatalog)) {
			// Move the products from source cart to target cart
			multiCartData = moveEntriesFromSourceCartToTargetCart(sourceCart, targetCart, multiCartData);
			// CLOUD-49955 CLOUD-54893
			status = multiCartData.getSuccessMessage();
			final String entryRemovedStatus;
			// Remove the selected entries from the source cart after moving the products to the target cart
			if (StringUtils.isNotBlank(status) && RexelConstants.STATUS_SUCCESS.equalsIgnoreCase(status)) {
				if (BooleanUtils.isTrue(multiCartData.getIsEntriesMoveToNewCart())) {
					entryRemovedStatus = removeSelectedEntriesFromSourceCart(sourceCart, multiCartData);
				} else {
					final int[] entryNumbers = getEntryNumbersToRemove(multiCartData);
					entryRemovedStatus = rexelB2BCartFacade.removeMultipleEntriesFromCart(entryNumbers);
				}
				// Set the session cart after removing the selected entries from the source cart
				if (RexelConstants.STATUS_SUCCESS.equalsIgnoreCase(entryRemovedStatus)) {
					cartService.setSessionCart(targetCart);
					return RexelConstants.STATUS_SUCCESS;
				}
			}
		}
		return multiCartData.getErrorMessage();// CLOUD-49955 CLOUD-54893
	}

	private List<AbstractOrderEntryModel> validateQuoteProducts(List<AbstractOrderEntryModel> toMoveEntries,
			CartModel targetCart, MultiCartData multiCartData) {

		final boolean addMultipleQuotesConfig = rexelConfigService
				.getConfigBooleanValue(RexelConfigConstants.MULTIPLE_QUOTE_ITEMS, false);
		final boolean quoteLevelERPConfig = rexelConfigService
				.getConfigBooleanValue(RexelConfigConstants.QUOTE_LEVEL_ERP_ENABLE, false);
		if (BooleanUtils.isTrue(quoteLevelERPConfig)) {
			boolean isSourceHasPartiallyFixed = false;
			boolean isSourceHasFlexible = false;
			boolean isSourceHasCatalogueProduct = false;
			boolean isTargetHasPartiallyFixed = false;
			boolean isTargetHasCatalogueProduct = false;
			boolean isTargetHasFlexible = false;

			for (final AbstractOrderEntryModel entry : toMoveEntries) {
				final String quoteRuleType = entry.getQuoteRuleTypeForProduct();
				if (RexelConstants.PARTIALLY_FIXED.equalsIgnoreCase(quoteRuleType)) {
					isSourceHasPartiallyFixed = true;
				} else if (StringUtils.isBlank(quoteRuleType) || StringUtils.isBlank(entry.getQuotationNumber())) {
					isSourceHasCatalogueProduct = true;
				} else if (RexelConstants.FLEXIBLE.equalsIgnoreCase(quoteRuleType)) {
					isSourceHasFlexible = true;
				}
			}
			if (CollectionUtils.isEmpty(targetCart.getEntries())) {
				isTargetHasCatalogueProduct = true;
			}
			for(final AbstractOrderEntryModel entry : targetCart.getEntries()) {
	            String quoteRuleType = entry.getQuoteRuleTypeForProduct();
	            if (RexelConstants.PARTIALLY_FIXED.equalsIgnoreCase(quoteRuleType)) {
	                isTargetHasPartiallyFixed = true;
	            } else if (StringUtils.isBlank(quoteRuleType) || StringUtils.isBlank(entry.getQuotationNumber())) {
	            	isTargetHasCatalogueProduct = true;
	            } else if (RexelConstants.FLEXIBLE.equalsIgnoreCase(quoteRuleType)) {
	            	isTargetHasFlexible = true;
				}
	        }
			if (isSourceHasPartiallyFixed && isSourceHasCatalogueProduct && (!isTargetHasCatalogueProduct || isTargetHasFlexible)) {
				setMultiCartDataError(multiCartData, toMoveEntries, RexelConstants.PARTIALLY_FIXED, RexelConstants.PARTIALLY_FIXED_QUOTE_EXISTS);
			}
			else if (isSourceHasFlexible && isSourceHasCatalogueProduct && isTargetHasPartiallyFixed) {
				setMultiCartDataError(multiCartData, toMoveEntries, RexelConstants.FLEXIBLE, RexelConstants.PARTIALLY_FIXED_QUOTE_EXISTS);
			}
			else if (isTargetHasPartiallyFixed && !isSourceHasCatalogueProduct) {
				clearEntriesAndSetErrorMessage(toMoveEntries,multiCartData ,RexelConstants.TARGET_HAS_PARTIALLY_FIXED_QUOTE);
				
			}
			else if (((isTargetHasPartiallyFixed && isSourceHasFlexible) || (isTargetHasFlexible && isSourceHasPartiallyFixed)) && (!isSourceHasCatalogueProduct || !isTargetHasCatalogueProduct)) {
			     clearEntriesAndSetErrorMessage(toMoveEntries, multiCartData, RexelConstants.TARGET_HAS_PARTIALLY_FIXED_QUOTE);
			}
			else if (StringUtils.isNotBlank(targetCart.getQuoteRuleType())
					&& targetCart.getQuoteRuleType().equalsIgnoreCase(RexelConstants.FIXED)) {
				clearEntriesAndSetErrorMessage(toMoveEntries,multiCartData,RexelConstants.FIXED_QUOTE_EXISTS );
			}
		} else if (BooleanUtils.isFalse(addMultipleQuotesConfig) && targetCart.getEntries().stream()
				.anyMatch(entry -> StringUtils.isNotBlank(entry.getQuotationNumber()))) {
			clearEntriesAndSetErrorMessage(toMoveEntries,multiCartData ,RexelConstants.FIXED_QUOTE_EXISTS);
		}
		return toMoveEntries;
	}

			boolean setCartNamePostDeletion = false;

			// remove the current cart from session
			sessionService.removeAttribute(SESSION_CART_PARAMETER_NAME);

			if (savedCartDataList != null && !savedCartDataList.isEmpty()) {
				isRestored = multiCartFacade.restoreSavedCart(savedCartDataList.get(0).getCartId());
			} else {
				// no saved carts exists for the user
				final String newCartName = multiCartService.getNewCartName(cart);
				setCartNamePostDeletion = newCartName.equals(
						RexelUtils.getMessageSource().getMessage(RexelConstants.DEFAULT_CART_NAME, null, loc))
								? false
								: true;
				MultiCartData multiCartData = new MultiCartData();
				multiCartData.setCartName(newCartName);
				isCreated = multiCartFacade.createNewSavedCart(multiCartData);
			}

			if (isRestored || isCreated) {
				if (isCreated && setCartNamePostDeletion) {
					multiCartFacade.setNewCartName(cartService.getSessionCart());
				}
			} else {
				// set the old cart to session
				cartService.setSessionCart(cart);
			}
		}
	}
	/**
	 * CLOUD-49957
	 * This method is used to check if the products are from the same catalog
	 * @param targetCart
	 * @param multiCartData
	 * @param movingProductOverCatalog
	 * throws FacetConfigServiceException, FacetSearchException
	 * @return boolean
	 */
	private boolean isProductsFromSameCatalog( final CartModel targetCart, MultiCartData multiCartData, boolean movingProductOverCatalog) throws FacetConfigServiceException, FacetSearchException {
		List<String> pcCatalogsList = new ArrayList<String>();
		Set<String> availableProductForPC = null;
		final RexelPersonalCatalogData rexelPersonalCatalogData = (RexelPersonalCatalogData) sessionService.getAttribute(
				RexelConstants.PERSONAL_CATALOG_ATTR);
		//CLOUD-56233
		final String currentPCCatalogName = rexelPersonalCatalogData !=null && rexelPersonalCatalogData.getDefaultPCProfile()!=null? rexelPersonalCatalogData.getDefaultPCProfile().getCode(): null;
		if (StringUtils.isNotBlank(currentPCCatalogName)) {
			pcCatalogsList.add(currentPCCatalogName);
			final Set<String> lCartProductCodes = new HashSet<String>();
			//CLOUD-55582
			if (CollectionUtils.isNotEmpty(targetCart.getEntries())){
				for (final AbstractOrderEntryModel orderEntry : targetCart.getEntries()) {
					String entry = orderEntry.getProduct().getCode();
					lCartProductCodes.add(entry);
				}
				LOG.info("Cart Product Codes: " + lCartProductCodes);
				if (CollectionUtils.isNotEmpty(pcCatalogsList) && CollectionUtils.isNotEmpty(lCartProductCodes)) {
					availableProductForPC = rexelSearchService.getPCProductCodesForCartItems(lCartProductCodes, pcCatalogsList);
				}
				LOG.info("Available Product for PC: " + availableProductForPC);
				//CLOUD-55585
				if (CollectionUtils.isNotEmpty(availableProductForPC) && lCartProductCodes.size() != 0) {
					//CLOUD-55636
					for (final String product : lCartProductCodes) {
                    		//CLOUD-56089
						if (!availableProductForPC.contains(product.toLowerCase()) && !NonCatalogProductUtility.isNonCatalogProduct(product)) {
							//CLOUD-55582
							multiCartData.setErrorMessage(RexelConstants.ERROR_FOR_PC);
							return false;
						}
					}

				}
			}
		}
		return movingProductOverCatalog;
	}

/**
 * Default implementation of the interface {@link SaveCartFacade}
 */
public class DefaultSaveCartFacade extends DefaultCartFacade implements SaveCartFacade
{
	private static final Logger LOG = Logger.getLogger(DefaultSaveCartFacade.class);

	private CommerceSaveCartService commerceSaveCartService;
	private CommerceSaveCartTextGenerationStrategy saveCartTextGenerationStrategy;
	private ConfigurationService configurationService;

	@Override
	public CommerceSaveCartResultData saveCart(final CommerceSaveCartParameterData inputParameters)
			throws CommerceSaveCartException
	{
		final CommerceSaveCartParameter parameter = new CommerceSaveCartParameter();

		if (StringUtils.isEmpty(inputParameters.getCartId()))
		{
			parameter.setCart(getCartService().getSessionCart());
		}
		else
		{
			final CartModel cartForCodeAndUser = getCommerceCartService().getCartForCodeAndUser(inputParameters.getCartId(),
					getUserService().getCurrentUser());

			if (cartForCodeAndUser == null)
			{
				throw new CommerceSaveCartException("Cannot find a cart for code [" + inputParameters.getCartId() + "]");
			}
			parameter.setCart(cartForCodeAndUser);
		}

		parameter.setEnableHooks(inputParameters.isEnableHooks());
		parameter.setName(generateSaveCartName(parameter.getCart(), inputParameters.getName()));
		parameter.setDescription(generateSaveCartDescription(parameter.getCart(), inputParameters.getDescription()));

		final CommerceSaveCartResult saveCartResult = getCommerceSaveCartService().saveCart(parameter);

		final CommerceSaveCartResultData saveCartResultData = new CommerceSaveCartResultData();
		saveCartResultData.setSavedCartData(getCartConverter().convert(saveCartResult.getSavedCart()));
		return saveCartResultData;
	}

	@Override
	public CommerceSaveCartResultData flagForDeletion(final String cartId) throws CommerceSaveCartException
	{
		final CommerceSaveCartParameter parameter = new CommerceSaveCartParameter();

		if (StringUtils.isEmpty(cartId))
		{
			throw new CommerceSaveCartException("No parameters defined");
		}
		else
		{
			final CartModel cartForCodeAndUser = getCommerceCartService().getCartForCodeAndUser(cartId,
					getUserService().getCurrentUser());

			if (cartForCodeAndUser == null)
			{
				throw new CommerceSaveCartException("Cannot find a cart for code [" + cartId + "]");
			}
			parameter.setCart(cartForCodeAndUser);
			parameter.setEnableHooks(getConfigurationService().getConfiguration().getBoolean(
					CommerceServicesConstants.FLAGFORDELETIONHOOK_ENABLED, true));
		}

		final CommerceSaveCartResult saveCartResult = getCommerceSaveCartService().flagForDeletion(parameter);

		final CommerceSaveCartResultData flagForDeletionResultData = new CommerceSaveCartResultData();
		flagForDeletionResultData.setSavedCartData(getCartConverter().convert(saveCartResult.getSavedCart()));

		return flagForDeletionResultData;
	}

	@Override
	public CartRestorationData restoreSavedCart(final CommerceSaveCartParameterData parameters) throws CommerceSaveCartException
	{
		final CommerceSaveCartParameter parameter = new CommerceSaveCartParameter();

		final CartModel cartForCodeAndUser = getCommerceCartService().getCartForCodeAndUser(parameters.getCartId(),
				getUserService().getCurrentUser());

		if (null == cartForCodeAndUser)
		{
			throw new CommerceSaveCartException("Cannot find a cart for code [" + parameters.getCartId() + "]");
		}

		parameter.setCart(cartForCodeAndUser);
		parameter.setEnableHooks(parameters.isEnableHooks());

		return getCartRestorationConverter().convert(getCommerceSaveCartService().restoreSavedCart(parameter));
	}

	@Override
	public CommerceSaveCartResultData getCartForCodeAndCurrentUser(final CommerceSaveCartParameterData inputParameters)
			throws CommerceSaveCartException
	{
		CartModel cartToRetrieve = null;

		if (StringUtils.isNotEmpty(inputParameters.getCartId()))
		{
			cartToRetrieve = getCommerceCartService().getCartForCodeAndUser(inputParameters.getCartId(),
					getUserService().getCurrentUser());
		}
		else
		{
			throw new CommerceSaveCartException("Cart code cannot be empty");
		}

		if (null == cartToRetrieve)
		{
			throw new CommerceSaveCartException("Cannot find a cart for code [" + inputParameters.getCartId() + "]");
		}

		final CommerceSaveCartResultData saveCartResultData = new CommerceSaveCartResultData();
		saveCartResultData.setSavedCartData(getCartConverter().convert(cartToRetrieve));

		return saveCartResultData;
	}

	@Override
	public SearchPageData<CartData> getSavedCartsForCurrentUser(final PageableData pageableData,
			final List<OrderStatus> orderStatus)
	{
		final SearchPageData<CartData> result = new SearchPageData<>();
		final SearchPageData<CartModel> savedCartModels = getCommerceSaveCartService().getSavedCartsForSiteAndUser(pageableData,
				getBaseSiteService().getCurrentBaseSite(), getUserService().getCurrentUser(), orderStatus);

		result.setPagination(savedCartModels.getPagination());
		result.setSorts(savedCartModels.getSorts());

		final List<CartData> savedCartDatas = Converters.convertAll(savedCartModels.getResults(), getCartConverter());

		result.setResults(savedCartDatas);
		return result;
	}

	@Override
	public Integer getSavedCartsCountForCurrentUser()
	{
		return getCommerceSaveCartService().getSavedCartsCountForSiteAndUser(getBaseSiteService().getCurrentBaseSite(),
				getUserService().getCurrentUser());
	}

	@Override
	public CommerceSaveCartResultData cloneSavedCart(final CommerceSaveCartParameterData parameter)
			throws CommerceSaveCartException
	{
		if (StringUtils.isEmpty(parameter.getCartId()))
		{
			throw new CommerceSaveCartException("Cart code cannot be empty");
		}

		final CommerceSaveCartParameter commerceSaveCartParameter = new CommerceSaveCartParameter();
		final CartModel cart = getCommerceCartService().getCartForCodeAndUser(parameter.getCartId(),
				getUserService().getCurrentUser());

		if (cart == null || cart.getSaveTime() == null)
		{
			throw new CommerceSaveCartException("Cannot find a saved cart for code [" + parameter.getCartId() + "]");
		}

		commerceSaveCartParameter.setCart(cart);

		commerceSaveCartParameter.setName(generateSaveCartName(cart, parameter.getName(),
				StringUtils.isEmpty(parameter.getName()) ? true : false));
		commerceSaveCartParameter.setDescription(generateSaveCartDescription(cart, parameter.getDescription()));
		commerceSaveCartParameter.setEnableHooks(parameter.isEnableHooks());

		final CommerceSaveCartResult saveCartResult = getCommerceSaveCartService().cloneSavedCart(commerceSaveCartParameter);
		final CommerceSaveCartResultData saveCartResultData = new CommerceSaveCartResultData();

		saveCartResultData.setSavedCartData(getCartConverter().convert(saveCartResult.getSavedCart()));

		return saveCartResultData;
	}

	protected String generateSaveCartName(final CartModel cartModel, final String name)
	{
		return generateSaveCartName(cartModel, name, false);
	}

	/**
	 * When restoring a saved cart, one copy of the restored saved cart can be kept. The name of the copied/(cloned) cart
	 * is the original saved cart name + copy#. The property commerceservices.clone.savedcart.name.regex set the regex
	 * for the name suffix of #. The property is now set in project.properties of storefronts that require this enhanced saved cart cloning name.
	 */
	protected String generateSaveCartName(final CartModel cartModel, final String name, final boolean clone)
	{
		final String baseCartName = StringUtils.trim(cartModel.getName());
		if (clone && StringUtils.isNotEmpty(baseCartName))
		{
			final String copyCountRegex = getConfigurationService().getConfiguration().getString(
					"commerceservices.clone.savecart.name.regex"+"."+getBaseSiteService().getCurrentBaseSite().getUid());
			if (StringUtils.isNotEmpty(copyCountRegex))
			{
				return getSaveCartTextGenerationStrategy().generateCloneSaveCartName(cartModel, copyCountRegex);
			}
		}

		final String clonePrefix = Localization.getLocalizedString("commerceservices.cart.copyof");
		final StringBuffer nameBuffer = clone ? new StringBuffer(clonePrefix).append(" ") : new StringBuffer();

		if (StringUtils.isNotEmpty(name))
		{
			return nameBuffer.append(name).toString();
		}
		else if (StringUtils.isNotEmpty(baseCartName))
		{
			return nameBuffer.append(baseCartName).toString();
		}

		return getSaveCartTextGenerationStrategy().generateSaveCartName(cartModel);
	}

	protected String generateSaveCartDescription(final CartModel cartModel, final String description)
	{
		if (StringUtils.isNotEmpty(description))
		{
			return description;
		}
		else if (StringUtils.isNotEmpty(cartModel.getDescription()))
		{
			return cartModel.getDescription();
		}

		return getSaveCartTextGenerationStrategy().generateSaveCartDescription(cartModel);
	}

	protected CommerceSaveCartService getCommerceSaveCartService()
	{
		return commerceSaveCartService;
	}

	@Required
	public void setCommerceSaveCartService(final CommerceSaveCartService commerceSaveCartService)
	{
		this.commerceSaveCartService = commerceSaveCartService;
	}

	protected CommerceSaveCartTextGenerationStrategy getSaveCartTextGenerationStrategy()
	{
		return saveCartTextGenerationStrategy;
	}

	@Required
	public void setSaveCartTextGenerationStrategy(final CommerceSaveCartTextGenerationStrategy saveCartTextGenerationStrategy)
	{
		this.saveCartTextGenerationStrategy = saveCartTextGenerationStrategy;
	}

	protected ConfigurationService getConfigurationService()
	{
		return configurationService;
	}

	@Required
	public void setConfigurationService(final ConfigurationService configurationService)
	{
		this.configurationService = configurationService;
	}
